// 声明节点
struct ASTEnumDeclData { name: &byte, variants: & &ASTNode, variant_count: i32, methods: & &ASTNode, method_count: i32, is_pub: i32 }
struct ASTErrorDeclData { name: &byte, is_pub: i32 }
struct ASTInterfaceDeclData { name: &byte, methods: & &ASTNode, method_count: i32, is_pub: i32 }
struct ASTStructDeclData { name: &byte, fields: & &ASTNode, field_count: i32, methods: & &ASTNode, method_count: i32, type_params: & &ASTNode, type_param_count: i32, is_pub: i32, is_packed: i32 }
struct ASTUnionDeclData { name: &byte, variants: & &ASTNode, variant_count: i32, methods: & &ASTNode, method_count: i32, is_pub: i32 }
struct ASTMethodBlockData { type_name: &byte, methods: & &ASTNode, method_count: i32 }
struct ASTFnDeclData { name: &byte, params: & &ASTNode, param_count: i32, return_type: &ASTNode, body: &ASTNode, is_pub: i32, is_extern: i32, extern_lib: &byte, type_params: & &ASTNode, type_param_count: i32, is_variadic: i32, is_export: i32, export_name: &byte }
struct ASTMacroDeclData { name: &byte, params: & &ASTNode, param_count: i32, return_tag: &ASTNode, body: &ASTNode, is_pub: i32 }
struct ASTTypeAliasData { name: &byte, target_type: &ASTNode, is_pub: i32 }
struct ASTVarDeclData { name: &byte, var_type: &ASTNode, init_expr: &ASTNode, is_const: i32, is_pub: i32, is_static: i32, is_threadlocal: i32 }
struct ASTExternVarDeclData { name: &byte, var_type: &ASTNode, is_const: i32, is_extern: i32, is_export: i32, export_name: &byte, init_expr: &ASTNode }
struct ASTDestructureDeclData { names: & &byte, name_count: i32, init_expr: &ASTNode, is_const: i32 }
struct ASTUseStmtData { path: &byte, item: &byte, alias: &byte, is_pub: i32 }

// 语句节点
struct ASTIfStmtData { condition: &ASTNode, then_branch: &ASTNode, else_branch: &ASTNode }
struct ASTWhileStmtData { condition: &ASTNode, body: &ASTNode }
struct ASTForStmtData { var_name: &byte, iter_expr: &ASTNode, body: &ASTNode, is_ref: i32, range_start: &ASTNode, range_end: &ASTNode }
struct ASTBreakStmtData { }
struct ASTContinueStmtData { }
struct ASTReturnStmtData { expr: &ASTNode }
struct ASTDeferStmtData { body: &ASTNode }
struct ASTErrdeferStmtData { body: &ASTNode }
struct ASTTestStmtData { description: &byte, body: &ASTNode }

// 表达式节点
struct ASTAssignData { dest: &ASTNode, src: &ASTNode }
struct ASTBlockData { stmts: & &ASTNode, stmt_count: i32 }
struct ASTBinaryExprData { op: i32, left: &ASTNode, right: &ASTNode }
struct ASTUnaryExprData { op: i32, operand: &ASTNode }
struct ASTCallExprData { func: &ASTNode, args: & &ASTNode, arg_count: i32 }
struct ASTMemberAccessData { object: &ASTNode, member_name: &byte, is_ptr_access: i32 }
struct ASTArrayAccessData { array: &ASTNode, index: &ASTNode }
struct ASTSliceExprData { base: &ASTNode, start: &ASTNode, len: &ASTNode }
struct ASTStructInitData { type_name: &byte, fields: & &ASTNode, field_count: i32 }
struct ASTArrayLiteralData { elements: & &ASTNode, element_count: i32 }
struct ASTTupleLiteralData { elements: & &ASTNode, element_count: i32 }
struct ASTSizeofData { target_type: &ASTNode }
struct ASTLenData { target: &ASTNode }
struct ASTAlignofData { target_type: &ASTNode }
struct ASTCastExprData { expr: &ASTNode, target_type: &ASTNode }
struct ASTUnderscoreData { }
struct ASTStringInterpData { segments: &ASTStringInterpSegment, segment_count: i32 }
struct ASTParamsData { }
struct ASTTryExprData { expr: &ASTNode }
struct ASTCatchExprData { expr: &ASTNode, error_var: &byte, body: &ASTNode }
struct ASTMatchExprData { match_expr: &ASTNode, cases: & &ASTNode, case_count: i32, else_body: &ASTNode }

// 宏相关节点
struct ASTMCEvalData { expr: &ASTNode }
struct ASTMCCodeData { expr: &ASTNode }
struct ASTMCAstData { expr: &ASTNode }
struct ASTMCErrData { msg: &ASTNode }
struct ASTMCIinterpData { expr: &ASTNode }
struct ASTMCTypeData { expr: &ASTNode }

// 内置函数节点
struct ASTAwaitExprData { expr: &ASTNode }
struct ASTSrcNameData { }
struct ASTSrcPathData { }
struct ASTSrcLineData { }
struct ASTSrcColData { }
struct ASTFuncNameData { }
struct ASTSyscallData { number: &ASTNode, args: & &ASTNode, arg_count: i32 }
struct ASTPtrFromUsizeData { value: &ASTNode }
struct ASTUsizeFromPtrData { ptr: &ASTNode }
struct ASTVaStartData { ap: &ASTNode, last_param: &ASTNode }
struct ASTVaEndData { ap: &ASTNode }
struct ASTVaArgData { ap: &ASTNode, arg_type: &ASTNode }

// 类型节点
struct ASTTypeNamedData { name: &byte, type_args: & &ASTNode, type_arg_count: i32 }
struct ASTTypePointerData { pointed_type: &ASTNode, is_ffi_pointer: i32, is_const: i32 }

struct ASTTypeSliceData { element_type: &ASTNode, size_expr: &ASTNode }
struct ASTTypeTupleData { element_types: & &ASTNode, element_count: i32 }
struct ASTTypeErrorUnionData { payload_type: &ASTNode }
struct ASTTypeAtomicData { base_type: &ASTNode }
